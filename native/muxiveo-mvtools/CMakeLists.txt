cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(muxiveo-mvtools VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MUXIVEO_MVTOOLS_VS_INCLUDE "" CACHE PATH "En-têtes de VapourSynth R80")
find_package(Threads REQUIRED)
add_executable(muxiveo-mvtools src/main.cpp src/y4m.cpp)
target_include_directories(muxiveo-mvtools PRIVATE "${MUXIVEO_MVTOOLS_VS_INCLUDE}")
target_compile_definitions(muxiveo-mvtools PRIVATE MUXIVEO_MVTOOLS_VERSION="${PROJECT_VERSION}")
target_link_libraries(muxiveo-mvtools PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(muxiveo-mvtools PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
elseif(NOT APPLE)
  target_link_options(muxiveo-mvtools PRIVATE -static-libstdc++ -static-libgcc)
endif()
install(TARGETS muxiveo-mvtools RUNTIME DESTINATION .)
